# Notification-local carrier construction followed by the actual dependent
# retirement controller. Full outer notification propagation stays separate.
import Base
import ../representation/records.bend as R
import ../representation/base.bend as B
import ../runtime/pending.bend as Q
import ../runtime/constants.bend as K
import ./pending-notification-carrier.bend as Notify
import ./pending-notification-fixture.bend as Fixture
import ./pending-notification-storage.bend as Storage
import ./pending-one-edge-transport.bend as Transport
import ./pending-carrier-state.bend as State
import ./pending-retirement-controller.bend as Retire
import ./pending-tail-status.bend as Tail
import ./pending-settlement-frame.bend as Frame
import ./committed-storage-frame.bend as Content
import ./memo-render-fixture.bend as Values


def after(+w: R.World,+node: Nat,+source: Nat,+serial: Nat,tasks: List<&2,Q.Task>,visited: List<&2,Nat>,released: List<&2,Nat>,scheduled: Bool,root: Nat) -> Q.Machine:
  Q.visitRemoved(Notify.notified(w,node,source,serial),tasks,visited,released,scheduled,node,root,1n+source)

def cleared(+w: R.World,+node: Nat,+source: Nat,+serial: Nat,+tasks: List<&2,Q.Task>,+visited: List<&2,Nat>,+released: List<&2,Nat>,+scheduled: Bool,+root: Nat,snapshot: State.CarrierSnapshot<w,node,source>,noError: {B.has(R.Node.status(B.node(w,node)),K.STATUS_ERROR()) == False{} : Bool},valid: {B.valid(w,node) == True{} : Bool}) -> {B.has(R.Node.status(B.node(Tail.world(after(w,node,source,serial,tasks,visited,released,scheduled,root)),node)),K.STATUS_PENDING()) == False{} : Bool}:
  +next = Notify.notified(w,node,source,serial)
  Retire.removedCleared(next,tasks,visited,released,scheduled,node,root,source,Notify.single(w,node,source,serial,snapshot,noError),Transport.valid(w,next,node,Storage.allStored(w,node,source,serial),valid))

def content(+w: R.World,+node: Nat,+source: Nat,+serial: Nat,+tasks: List<&2,Q.Task>,+visited: List<&2,Nat>,+released: List<&2,Nat>,+scheduled: Bool,+root: Nat,noError: {B.has(R.Node.status(B.node(w,node)),K.STATUS_ERROR()) == False{} : Bool}) -> {Content.observe(Tail.world(after(w,node,source,serial,tasks,visited,released,scheduled,root))) == Content.observe(w) : Content.View}:
  +next = Notify.notified(w,node,source,serial)
  Content.compose(w,next,Tail.world(after(w,node,source,serial,tasks,visited,released,scheduled,root)),Frame.removed(next,tasks,visited,released,scheduled,node,root,1n+source),Notify.content(w,node,source,serial,noError))

def constructed(+values: List<&2,R.Value>,+value: R.Value,+extensions: List<&2,R.NodeExtension>,+sets: List<&2,List<&2,Nat>>,+incoming: Fixture.Incoming,+serial: Nat,tasks: List<&2,Q.Task>,visited: List<&2,Nat>,released: List<&2,Nat>,scheduled: Bool) -> Q.Machine:
  after(Fixture.initial(values,value,extensions,sets,incoming),Fixture.leaf(values,value,extensions,sets),B.length(R.Node,Values.nodes(values)),serial,tasks,visited,released,scheduled,Fixture.root(values))

def constructedCleared(+values: List<&2,R.Value>,+value: R.Value,+extensions: List<&2,R.NodeExtension>,+sets: List<&2,List<&2,Nat>>,+incoming: Fixture.Incoming,+serial: Nat,+tasks: List<&2,Q.Task>,+visited: List<&2,Nat>,+released: List<&2,Nat>,+scheduled: Bool) -> {B.has(R.Node.status(B.node(Tail.world(constructed(values,value,extensions,sets,incoming,serial,tasks,visited,released,scheduled)),Fixture.leaf(values,value,extensions,sets))),K.STATUS_PENDING()) == False{} : Bool}:
  cleared(Fixture.initial(values,value,extensions,sets,incoming),Fixture.leaf(values,value,extensions,sets),B.length(R.Node,Values.nodes(values)),serial,tasks,visited,released,scheduled,Fixture.root(values),Fixture.seed(values,value,extensions,sets,incoming),Fixture.noError(values,value,extensions,sets,incoming),Fixture.valid(values,value,extensions,sets,incoming))

def constructedContent(+values: List<&2,R.Value>,+value: R.Value,+extensions: List<&2,R.NodeExtension>,+sets: List<&2,List<&2,Nat>>,+incoming: Fixture.Incoming,+serial: Nat,+tasks: List<&2,Q.Task>,+visited: List<&2,Nat>,+released: List<&2,Nat>,+scheduled: Bool) -> {Content.observe(Tail.world(constructed(values,value,extensions,sets,incoming,serial,tasks,visited,released,scheduled))) == Content.observe(Fixture.initial(values,value,extensions,sets,incoming)) : Content.View}:
  content(Fixture.initial(values,value,extensions,sets,incoming),Fixture.leaf(values,value,extensions,sets),B.length(R.Node,Values.nodes(values)),serial,tasks,visited,released,scheduled,Fixture.root(values),Fixture.noError(values,value,extensions,sets,incoming))
